  group: 'API Reference',

  navTitle: 'Cinder.feedback',
};

Use `feedback()` when you want reflex output without invoking speech or the full `run()` pipeline.

## Example

```ts
const intent = Cinder.resolve('show weather');
const dispatch = await Cinder.dispatch(intent, document);
const reflex = Cinder.feedback(intent, dispatch);
console.log(reflex);
```
